# Direct Port 25 Socket SMTP Handshake Sequence
def verify_email_smtp(email, mx_host):
    try:
        server = smtplib.SMTP(timeout=10)
        server.connect(mx_host, 25)
        server.helo("fomins.co.in")
        server.mail("verify@fomins.co.in")
        code, message = server.rcpt(email)
        server.quit()
        # 250 OK indicates mailbox is active and deliverable
        return code == 250
    except Exception as e:
        return False
